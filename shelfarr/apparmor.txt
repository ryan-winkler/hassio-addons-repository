#include <tunables/global>

profile hassio-addons/shelfarr flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/bash>
  #include <abstractions/nameservice>
  #include <abstractions/openssl>

  # Baseline profile for Shelfarr in HAOS. Keep broad compatibility while
  # denying known high-risk kernel interfaces.
  file,
  network,
  # Capabilities required for ownership management and privilege drop (gosu).
  capability chown,
  capability dac_override,
  capability fowner,
  capability setgid,
  capability setuid,

  deny /proc/kcore rwklx,
  deny /proc/sysrq-trigger rwklx,
  deny /sys/firmware/** rwklx,
}
